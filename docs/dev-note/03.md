Hướng này đúng với yêu cầu. Thay vì tạo class con cho từng tổ hợp hiệu ứng, bạn giữ một bộ luật gốc và bọc các hiệu ứng lên trên. Mỗi hiệu ứng là một file, nhận luật phía trước qua tham số next, rồi có thể đổi kết quả hoặc làm thêm việc trước và sau khi luật gốc chạy.

Lý do không dùng class: với Knight, ProtectedKnight, ReviveKnight, ProtectedReviveKnight, số lớp tăng theo mọi tổ hợp. Ghép lớp thì mỗi hiệu ứng độc lập, muốn kết hợp thế nào cũng được.

Mình đặt tên theo camelCase cho đúng chuẩn JavaScript. chess_expansion tương ứng với kết quả của buildRules(chessBase, extensions).

core/chess.js (ví dụ chỉ viết quân Mã, các quân khác làm tương tự trong getMoves):

js
import { pieceAt, updatePiece } from './state.js'

const KNIGHT_JUMPS = [[-2, -1], [-2, 1], [-1, -2], [-1, 2], [1, -2], [1, 2], [2, -1], [2, 1]]
let seq = 0

export function createPiece({ type, color, row, col }) {
  return { id: `p${++seq}`, type, color, row, col, ext: {} }
}

export function createInitialState() {
  return {
    size: 8,
    turn: 'w',
    turnCount: 1,
    pieces: [
      createPiece({ type: 'N', color: 'w', row: 7, col: 1 }),
      createPiece({ type: 'N', color: 'b', row: 0, col: 6 }),
    ],
    graveyard: [],
    tiles: {},
  }
}

// Luật gốc. Nhận R là luật đã ghép, nên các hàm gọi nhau qua R
export function chessBase(R) {
  return {
    inside(state, row, col) {
      return row >= 0 && col >= 0 && row < state.size && col < state.size
    },

    getMoves(state, piece) {
      if (piece.type !== 'N') return []
      return KNIGHT_JUMPS
        .map(([dr, dc]) => ({ row: piece.row + dr, col: piece.col + dc }))
        .filter(({ row, col }) => R.inside(state, row, col))
    },

    isLegalMove(state, id, row, col) {
      const piece = state.pieces.find(p => p.id === id)
      if (!piece || piece.color !== state.turn) return false
      const reachable = R.getMoves(state, piece).some(m => m.row === row && m.col === col)
      const target = pieceAt(state, row, col)
      return reachable && (!target || target.color !== piece.color)
    },

    removePiece(state, row, col) {
      return { ...state, pieces: state.pieces.filter(p => !(p.row === row && p.col === col)) }
    },

    movePiece(state, id, row, col) {
      const cleared = R.removePiece(state, row, col) // ăn quân nếu có
      return updatePiece(cleared, id, { row, col })
    },

    endTurn(state) {
      return { ...state, turn: state.turn === 'w' ? 'b' : 'w', turnCount: state.turnCount + 1 }
    },
  }
}
4. Ghép luật: buildRules

core/compose.js:

js
// Trả về "chess_expansion": đủ hàm của luật gốc, hàm mới và hàm đã được ghi đè
export function buildRules(baseFactory, extensions) {
  const rules = {}
  const base = baseFactory(rules) // base gọi R.xxx nên luôn nhận bản mới nhất

  // Hàm mới của extension: gắn thẳng vào
  for (const ext of extensions) Object.assign(rules, ext.methods)

  // Hàm đã có: bọc từ trong ra ngoài, extension đầu danh sách nằm ngoài cùng
  for (const name of Object.keys(base)) {
    rules[name] = extensions.reduceRight((next, ext) => {
      const override = ext.override?.[name]
      return override ? (...args) => override(next, ...args) : next
    }, base[name])
  }
  return rules
}

Mỗi extension có hai phần: methods để thêm hàm mới, và override để đổi một hàm có sẵn. Tham số next là bản đứng sau trong chuỗi, và cuối cùng là luật gốc. Extension muốn dùng luật gốc chỉ cần gọi next(...), không cần copy logic.

Lỗi hay gặp nhất với kiểu này: nếu luật gốc gọi removePiece(...) trực tiếp thay vì R.removePiece(...), extension sẽ không chạy khi quân bị ăn. Vì vậy chessBase nhận R và luôn gọi qua R.

5. Extensions: bọc lên luật gốc

extensions/protect.js:

js
import { pieceAt, setExt } from '../core/state.js'

export const protect = {
  id: 'protect',

  methods: {
    grantProtect: (state, id, turns) => setExt(state, id, 'protect', { turnsLeft: turns }),
  },

  override: {
    // Luật gốc kiểm tra trước, rồi mới thêm luật bảo vệ
    isLegalMove(next, state, id, row, col) {
      if (!next(state, id, row, col)) return false
      const target = pieceAt(state, row, col)
      return !(target?.ext.protect?.turnsLeft > 0)
    },

    // Gọi next trước để lượt đổi xong, rồi mới trừ số lượt còn lại
    endTurn(next, state) {
      const s = next(state)
      return {
        ...s,
        pieces: s.pieces.map(p =>
          p.ext.protect
            ? { ...p, ext: { ...p.ext, protect: { turnsLeft: Math.max(0, p.ext.protect.turnsLeft - 1) } } }
            : p
        ),
      }
    },
  },
}

extensions/revive.js:

js
import { pieceAt, setExt } from '../core/state.js'

const REVIVE_DELAY = 2 // hồi sinh sau 2 lượt

export const revive = {
  id: 'revive',

  methods: {
    grantRevive: (state, id) => setExt(state, id, 'revive', { charges: 1 }),
  },

  override: {
    // Quân bị ăn: lưu lại quân (kèm vị trí và ext) vào mộ trước khi luật gốc xóa nó
    removePiece(next, state, row, col) {
      const target = pieceAt(state, row, col)
      const s = next(state, row, col)
      if (target?.ext.revive?.charges > 0) {
        return { ...s, graveyard: [...s.graveyard, { piece: target, diedAt: state.turnCount }] }
      }
      return s
    },

    // Cuối lượt: đủ thời gian và ô cũ còn trống thì đặt quân về đúng vị trí đó
    endTurn(next, state) {
      let s = next(state)
      const stillDead = []
      for (const entry of s.graveyard) {
        const { piece, diedAt } = entry
        const ready = s.turnCount - diedAt >= REVIVE_DELAY
        if (ready && !pieceAt(s, piece.row, piece.col)) {
          const charges = piece.ext.revive.charges - 1
          s = { ...s, pieces: [...s.pieces, { ...piece, ext: { ...piece.ext, revive: { charges } } }] }
        } else {
          stillDead.push(entry)
        }
      }
      return { ...s, graveyard: stillDead }
    },
  },
}

Room.js: một room có một bàn cờ, một bộ luật và tùy chọn một bot.

js
import { chessBase, createInitialState } from './core/chess.js'
import { buildRules } from './core/compose.js'
import { CARDS } from './cards/cardDefs.js'

export class Room {
  listeners = new Set()

  constructor(extensions = [], bot = null) {
    this.rules = buildRules(chessBase, extensions)
    this.state = createInitialState()
    this.bot = bot
  }

  // Điểm vào duy nhất: người chơi và bot đều gọi hàm này
  dispatch = (action) => {
    if (this.state.winner) return
    const next = this.reduce(action)
    if (next === this.state) return // thao tác không hợp lệ
    this.state = next
    this.notify()

    if (this.bot && this.state.turn === this.bot.color) {
      setTimeout(() => {
        const botAction = this.bot.choose(this.state, this.rules)
        if (botAction) this.dispatch(botAction)
      }, 500)
    }
  }

  reduce(action) {
    const R = this.rules
    const s = this.state
    switch (action.type) {
      case 'MOVE':
        if (!R.isLegalMove(s, action.id, action.row, action.col)) return s
        return R.endTurn(R.movePiece(s, action.id, action.row, action.col))
      case 'CARD': {
        const card = CARDS[action.cardId]
        return R.endTurn(R[card.effect](s, action.pieceId, ...card.args))
      }
      default:
        return s
    }
  }

  getState = () => this.state
  subscribe = (listener) => {
    this.listeners.add(listener)
    return () => this.listeners.delete(listener)
  }
  notify() {
    this.listeners.forEach(l => l())
  }
}

bot/simpleBot.js: bot dùng đúng API như người chơi.

js
export const simpleBot = {
  color: 'b',

  choose(state, R) {
    // Kỹ năng của bot, tạm thời là thẻ mở rộng bàn cờ
    if (Math.random() < 0.1) return { type: 'CARD', cardId: 'expand' }

    const moves = state.pieces
      .filter(p => p.color === this.color)
      .flatMap(p => R.getMoves(state, p).map(m => ({ type: 'MOVE', id: p.id, row: m.row, col: m.col })))
      .filter(a => R.isLegalMove(state, a.id, a.row, a.col))

    return moves[Math.floor(Math.random() * moves.length)] ?? null
  },
}

Vì bot và người chơi dùng chung dispatch, luật không bị viết hai lần. Sau này khi làm online, server chỉ cần chạy đúng class Room này.
