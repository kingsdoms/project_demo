src/games/chess/
├── index.js              # manifest cho vỏ: ghép core, engine, extensions
│
├── core/                 # CHỈ luật cờ vua cơ bản. Không biết Room, extension, UI, bot
│   ├── board.js          # bàn: kích thước, danh sách quân, ô có nằm trong bàn không
│   ├── piece.js          # quân: dữ liệu quân và bảng loại quân (PIECE_DEFS)
│   ├── game.js           # ván: lượt, kết quả, nước vừa đi
│   ├── move.js           # bản ghi nước đi đã hợp lệ
│   ├── action.js         # ý định đầu vào, chưa kiểm tra
│   ├── movement.js       # hình học: quân đi và ăn được ô nào, chưa xét chiếu
│   └── rules.js          # luật: nước hợp lệ, chiếu, nhập thành, bắt qua đường, kết thúc
│
├── engine/               # máy chạy ván, không biết cờ vua
│   ├── compose.js        # buildRules: ghép luật gốc với extension
│   └── Room.js           # giữ game và rules, dispatch, subscribe, restart, destroy
│
├── extensions/           # chỉ import từ core/
├── bot/                  # đọc game và rules, trả về action
└── ui/                   # đọc game qua useRoom, gửi action




Hình dạng dữ liệu
js
// Quân
{ id: 'p12', type: 'N', color: 'w', row: 7, col: 1, hasMoved: false, ext: {} }
// ext là chỗ trống cho extension, core không đọc

// Action (ý định)
{ type: 'MOVE', pieceId: 'p12', to: { row: 5, col: 2 }, promotion: null }

// Move (bản ghi đã hợp lệ)
{ pieceId: 'p12', from: { row: 7, col: 1 }, to: { row: 5, col: 2 },
  captured: null, special: null, promotion: null }
// special: 'castle' | 'enPassant' | null

// Game (ván)
{ board: { size: 8, pieces: [...] }, turn: 'w', enPassant: null,
  lastMove: null, status: 'playing', winner: null }
Hàm của luật
js
// movement.js: không xét chiếu
movesOf(board, piece)        // ô quân có thể đi
attacksOf(board, piece)      // ô quân đang kiểm soát

// rules.js: luật đầy đủ, nhận game
legalMoves(game, piece)      // Move hợp lệ của một quân, đã lọc chiếu
allLegalMoves(game)          // dùng cho bot và kiểm tra hết nước
isInCheck(game, color)
applyMove(game, move)        // trả về game mới
status(game)                 // cập nhật status và winner
applyAction(game, action)    // điểm vào duy nhất
Luồng một nước đi
UI hoặc bot tạo Action
  └─► Room.dispatch(action)
        └─► rules.applyAction(game, action)
              ├─ kiểm tra: quân tồn tại, đúng màu, đúng lượt
              ├─ legalMoves(game, piece): tìm Move khớp với action
              ├─ applyMove(game, move): game mới, quân di chuyển, quân bị ăn, đổi lượt
              └─ status(game): cập nhật kết quả
        └─► Room lưu game mới và báo UI
