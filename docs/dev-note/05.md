└── ui/                   # đọc game qua useRoom, gửi action




Hình dạng dữ liệu
js
// Quân
{ id: 'p12', type: 'N', color: 'w', row: 7, col: 1, hasMoved: false, ext: {} }
// ext là chỗ trống cho extension, core không đọc

// Action (ý định)
{ type: 'MOVE', pieceId: 'p12', to: { row: 5, col: 2 }, promotion: null }

// Move (bản ghi đã hợp lệ)
{ pieceId: 'p12', from: { row: 7, col: 1 }, to: { row: 5, col: 2 },
  captured: null, special: null, promotion: null }
// special: 'castle' | 'enPassant' | null

// Game (ván)
{ board: { size: 8, pieces: [...] }, turn: 'w', enPassant: null,
  lastMove: null, status: 'playing', winner: null }
Hàm của luật
js
// movement.js: không xét chiếu
movesOf(board, piece)        // ô quân có thể đi
attacksOf(board, piece)      // ô quân đang kiểm soát

// rules.js: luật đầy đủ, nhận game
legalMoves(game, piece)      // Move hợp lệ của một quân, đã lọc chiếu
allLegalMoves(game)          // dùng cho bot và kiểm tra hết nước
isInCheck(game, color)
applyMove(game, move)        // trả về game mới
status(game)                 // cập nhật status và winner
applyAction(game, action)    // điểm vào duy nhất
Luồng một nước đi
UI hoặc bot tạo Action
  └─► Room.dispatch(action)
        └─► rules.applyAction(game, action)
              ├─ kiểm tra: quân tồn tại, đúng màu, đúng lượt
              ├─ legalMoves(game, piece): tìm Move khớp với action
              ├─ applyMove(game, move): game mới, quân di chuyển, quân bị ăn, đổi lượt
              └─ status(game): cập nhật kết quả
        └─► Room lưu game mới và báo UI

Nếu app tự xử lý thao tác, app phải hiểu «bấm ô khi đã chọn quân thì là đi, bấm quân mình thì là đổi quân chọn». Đó đã là kiến thức cờ vua. Vì vậy app nên lo phần nối dây và các việc chung (khóa input, bắt lỗi, âm thanh). Việc diễn giải thao tác thành nước đi nên nằm trong một controller/ của chính games/chess, không nằm ở ui và không nằm ở app.

Cấu trúc thư mục

games/chess/
├─ index.js        manifest của game, không import ui/
├─ core/           luật + room (đã có)
├─ controller/     giữ room, trạng thái chọn quân, actions, view-model
│   ├─ createController.js
│   ├─ interaction.js   clickSquare → chọn / đi / bỏ chọn (hàm thuần, test bằng node)
│   └─ toView.js        (game, interaction) → view
├─ bot/            bot được nối trong create(), app không thấy
├─ extensions/
└─ ui/
    ├─ index.js    export default { Board, Panel, Overlay }
    ├─ Board/      BoardView, SquareView, PieceView
    ├─ Panel/      StatusBar, MoveList

Manifest và controller (hợp đồng giữa app và game)

js
// games/chess/index.js
export default {
  id: 'chess',
  title: 'Cờ vua',
  create(options) { /* dựng Room(reduce), bot... rồi trả controller */ },
  loadUI: () => import('./ui/index.js'),   // lazy, để test/server không kéo ui
}

// controller trả về
{
  getView(),        // snapshot dữ liệu thuần để vẽ (trả cùng tham chiếu nếu chưa đổi)
  subscribe(fn),    // trả hàm hủy
  actions: { clickSquare(row, col), promote(type), restart() },
  onError(fn),      // nước sai (GameError) báo ở đây, không ném ra app
  destroy(),
}

view là dữ liệu thuần, ví dụ:

js
{ pieces: [{ id, type, side, row, col }], selectedId, targets: [{ row, col }],
  checkedKing, turn, promotion, result }

Ô đi được (targets) do controller tính bằng getLegalMoves rồi đưa vào view. Vì vậy ui không cần import core/ hay engine/ nữa.

App tải game

js
// app/catalog.js: nơi duy nhất app nhắc tên game
export const catalog = { chess: () => import('../games/chess/index.js') }

// app/startGame.js
async function startGame(id, options) {
  const game = (await catalog[id]()).default
  const controller = game.create(options)
  try {
    const ui = (await game.loadUI()).default
    mount({ controller, ui })
  } catch (e) { controller.destroy(); throw e }   // tải ui lỗi thì dọn controller
}

// app/GameScreen (viết kiểu React, Vue/Svelte giữ nguyên ý)
function GameScreen({ controller, ui }) {
  const view = useController(controller)          // subscribe + getView
  const actions = guard(controller.actions)       // bọc: khóa input, âm thanh, log
  controller.onError(showToast)
  return <>
    <ui.Board view={view} actions={actions} />
    {ui.Panel && <ui.Panel view={view} actions={actions} />}
    {ui.Overlay && <ui.Overlay view={view} actions={actions} />}
  </>
}

App chỉ biết ba slot Board, Panel, Overlay. Board bắt buộc, hai slot còn lại tùy game. Nhờ đó app giữ quyền bố cục (header, nút về menu, nút restart).

Luồng khi người dùng thao tác

Bấm "Bắt đầu"
  app: catalog.chess()      → import games/chess/index.js
  app: game.create()        → controller (room + bot + ...)
  app: game.loadUI()        → { Board, Panel, Overlay }
Bấm một ô
  ui:         actions.clickSquare(r, c)
  app:        guard(...) → controller.actions.clickSquare(r, c)
  controller: chọn quân, hoặc room.dispatch(Action.move(...))
  room → subscribe → app → view mới → ui vẽ lại

Quy tắc cho ui/, cùng kiểu với quy tắc của core/:

Chỉ nhận view và actions qua props. Không import core/, engine/, controller/.
Không giữ state của game. Chỉ có state thuần trình bày như hover hay animation.
Gọi actions.xxx để báo ý định, không tự dispatch.
Chỉ cần dựng một view giả là chạy được, nên xem trước và test component dễ.
